-- POLÍTICAS QUE VOY A IMPLEMENTAR

1) FUNCIONES AUXILIARES (SECURITY DEFINER)

CREATE OR REPLACE FUNCTION auth_es_admin()
RETURNS boolean AS $$
  SELECT EXISTS (
    SELECT 1 FROM public.perfiles WHERE id = auth.uid() AND rol = 'admin'
  );
$$ LANGUAGE sql SECURITY DEFINER;

CREATE OR REPLACE FUNCTION auth_es_staff()
RETURNS boolean AS $$
  SELECT EXISTS (
    SELECT 1 FROM public.perfiles WHERE id = auth.uid() AND rol IN ('operador', 'admin')
  );
$$ LANGUAGE sql SECURITY DEFINER;

-- TABLA: PERFILES

CREATE POLICY "Los usuarios pueden ver su propio perfil" ON perfiles
  FOR SELECT USING (auth.uid() = id);

CREATE POLICY "Los usuarios pueden actualizar su propio perfil" ON perfiles
  FOR UPDATE USING (auth.uid() = id);

CREATE POLICY "Admins pueden ver todo en perfiles" ON perfiles
  FOR SELECT USING ( auth_es_admin() );

CREATE POLICY "Admins pueden actualizar todo en perfiles" ON perfiles
  FOR UPDATE USING ( auth_es_admin() );

-- Permitimos insertar libremente para que el Trigger de Supabase no falle al crear la cuenta.
-- (La seguridad de que no inserten otros IDs está garantizada por la llave foránea de auth.users)
CREATE POLICY "Permitir insertar perfil" ON perfiles
  FOR INSERT WITH CHECK (true);



-- TABLA: BILLETERAS (usa: usuario_id)

-- Los usuarios necesitan poder ver, insertar y actualizar su saldo
CREATE POLICY "Usuarios gestionan su propia billetera" ON billeteras
  FOR ALL USING (usuario_id = auth.uid());

CREATE POLICY "Admins ven todas las billeteras" ON billeteras
  FOR ALL USING ( auth_es_admin() );



-- TABLA: METODOS_PAGO (usa: usuario_id)

CREATE POLICY "Usuarios gestionan sus propios métodos de pago" ON metodos_pago
  FOR ALL USING (usuario_id = auth.uid());

CREATE POLICY "Admins gestionan todos los métodos de pago" ON metodos_pago
  FOR ALL USING ( auth_es_admin() );



-- TABLA: GASOLINERAS

CREATE POLICY "Admins gestionan gasolineras" ON gasolineras

-- TABLA: CUPONES (usa: propietario_id)

-- TABLA: PAGOS_QR (usa: cliente_id, operador_id)

-- El cliente necesita poder generar (INSERT) y leer el QR, pero al escanear, el QR no tiene su ID todavía
CREATE POLICY "Todos los autenticados pueden leer los QR" ON pagos_qr
  FOR SELECT USING (auth.role() = 'authenticated');

CREATE POLICY "Clientes gestionan sus propios pagos QR" ON pagos_qr
  FOR ALL USING (cliente_id = auth.uid());

CREATE POLICY "Operadores y Admins gestionan pagos QR" ON pagos_qr
  FOR ALL USING ( auth_es_staff() );
